pub fn _packetout<'a>(
    os: &mut OggStreamState<'a>,
    op: Option<&mut OggPacket<'a>>,
    adv: i32,
) -> i32 {
    let mut ptr = os.lacing_returned as usize;
    if os.lacing_packet <= ptr as i64 {
        return 0;
    }
    if os.lacing_vals[ptr] & 0x400 != 0 {
        os.lacing_returned += 1;
        os.packetno += 1;
        return -1;
    }
    if op.is_none() && adv == 0 {
        return 1;
    }
    let mut size = os.lacing_vals[ptr] & 0xff;
    let mut bytes = size as i64;
    let mut eos = os.lacing_vals[ptr] & 0x200;
    let bos = os.lacing_vals[ptr] & 0x100;
    while size == 255 {
        ptr += 1;
        let val = os.lacing_vals[ptr];
        size = val & 0xff;
        if val & 0x200 != 0 {
            eos = 0x200;
        }
        bytes += size as i64;
    }
    if let Some(op) = op {
        op.e_o_s = eos as i64;
        op.b_o_s = bos as i64;
        op.packet = &os
            .body_data[os.body_returned as usize..(os.body_returned + bytes) as usize];
        op.packetno = os.packetno;
        op.granulepos = os.granule_vals[ptr];
        op.bytes = bytes;
    }
    if adv != 0 {
        os.body_returned += bytes;
        os.lacing_returned = ptr as i64 + 1;
        os.packetno += 1;
    }
    1
}
